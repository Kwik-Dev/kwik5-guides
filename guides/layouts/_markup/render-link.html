{{/* Link render hook. The guides are written for readers who will also open the planning docs, so a few links point at ../../decisions.md and friends. Those files are not part of this site, so instead of shipping a 404 they are rendered as a plain reference. Every other .md link is resolved to the real page URL — Hugo's content graph does the lookup, so a renamed guide breaks the build instead of the link. */}} {{- $dest := .Destination -}} {{- $text := .Text -}} {{- $title := .Title -}} {{- if findRE `^(https?:|mailto:|tel:)` $dest -}} {{ $text }} {{- else if hasPrefix $dest "#" -}} {{ $text }} {{- else -}} {{- $u := urls.Parse $dest -}} {{- with .Page.GetPage $u.Path -}} {{ $text }} {{- else -}} {{ $text }} {{- end -}} {{- end -}}
