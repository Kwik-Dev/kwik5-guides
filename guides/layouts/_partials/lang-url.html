Usage: {{ partial "lang-url.html" "credits" }} */}} {{- $path := . -}} {{- with site.GetPage $path -}}{{ return .RelPermalink }}{{- end -}} {{- range hugo.Sites -}} {{- with .GetPage $path -}}{{ return .RelPermalink }}{{- end -}} {{- end -}} {{- return (relLangURL (printf "%s/" $path)) -}}
